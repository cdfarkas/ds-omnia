### <Localized en="What Omnia requires" fr="Ce qu'Omnia exige" />

<Localized
  en={

`providers.tsx`

```tsx
'use client';

import { useState, type ReactNode } from 'react';
import { OmniaThemeProvider, type Theme } from '@cfarkas/omnia';

export function Providers({
  initialTheme,
  children,
}: {
  initialTheme: Theme;
  children: ReactNode;
}) {
  const [theme, setTheme] = useState(initialTheme);
  return (
    <OmniaThemeProvider
      theme={theme}
      skin="default"
      locale="en"
      onThemeChange={(next) => {
        document.cookie = `theme=${next}; path=/; max-age=31536000; samesite=lax`;
        setTheme(next);
      }}
    >
      {children}
    </OmniaThemeProvider>
  );
}
```

`app/layout.tsx`

```tsx
import '@cfarkas/omnia/styles.css';
import '@cfarkas/omnia/fonts.css';
import { cookies } from 'next/headers';
import type { ReactNode } from 'react';
import { Providers } from './providers';

export default async function RootLayout({ children }: { children: ReactNode }) {
  const theme = (await cookies()).get('theme')?.value === 'dark' ? 'dark' : 'light';
  return (
    <html lang="en" data-theme={theme} data-skin="default">
      <body>
        <Providers initialTheme={theme}>{children}</Providers>
      </body>
    </html>
  );
}
```

`app/root.tsx`

```tsx
import '@cfarkas/omnia/styles.css';
import '@cfarkas/omnia/fonts.css';
import type { ReactNode } from 'react';
import {
  Links,
  Meta,
  Outlet,
  Scripts,
  ScrollRestoration,
  useRouteLoaderData,
} from 'react-router';
import type { Theme } from '@cfarkas/omnia';
import { Providers } from './providers';

export function loader({ request }: { request: Request }) {
  const dark = /(?:^|;\s*)theme=dark(?:;|$)/.test(request.headers.get('cookie') ?? '');
  return { theme: (dark ? 'dark' : 'light') as Theme };
}

export function Layout({ children }: { children: ReactNode }) {
  const data = useRouteLoaderData<typeof loader>('root');
  return (
    <html lang="en" data-theme={data?.theme ?? 'light'} data-skin="default">
      <head>
        <meta charSet="utf-8" />
        <Meta />
        <Links />
      </head>
      <body>
        {children}
        <ScrollRestoration />
        <Scripts />
      </body>
    </html>
  );
}

export default function App({ loaderData }: { loaderData: { theme: Theme } }) {
  return (
    <Providers initialTheme={loaderData.theme}>
      <Outlet />
    </Providers>
  );
}
```
